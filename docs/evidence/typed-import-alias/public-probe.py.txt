import json,tempfile,subprocess,os
from pathlib import Path
D=Path('docs/evidence/typed-import-alias');p=json.loads((D/'protocol.json').read_text())
with tempfile.TemporaryDirectory() as d:
 root=Path(d)/'source';root.mkdir()
 for f,s in p['sources'].items():(root/f).write_text(s)
 def run(*args):
  r=subprocess.run(['python','tools/agent_memory.py',*args,'--project',str(root),'--memory-home',d+'/memory'],capture_output=True,text=True,check=True);return json.loads(r.stdout) if '--json'in args else r.stdout
 run('init');run('learn-path','--path',str(root))
 out={k:run('list','--type',k,'--limit','300','--json')for k in ['code-symbol','memory-edge']};out['contexts']={q:run('context','--query',q,'--json')for q in p['queries']}
 syms={s['id']:s['qualified_name']for s in out['code-symbol']};out['typed_calls']=[{'source':syms[e['source_id']],'target':syms[e['target_id']],'evidence':e['evidence']}for e in out['memory-edge']if e['source_type']=='code_symbol'and e['target_type']=='code_symbol'and e['relation']=='calls']
 (D/(os.environ.get('OUT','before')+'.json')).write_text(json.dumps(out,indent=2));print(out['typed_calls'])
