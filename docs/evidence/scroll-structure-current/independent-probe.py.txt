import json,subprocess,tempfile
from pathlib import Path
D=Path('docs/evidence/scroll-structure-current');p=json.loads((D/'protocol.json').read_text())
with tempfile.TemporaryDirectory() as d:
 root=Path(d)/'src';root.mkdir()
 for f,s in p['sources'].items():(root/f).write_text(s)
 def run(*args):
  r=subprocess.run(['python','tools/agent_memory.py',*args,'--project',str(root),'--memory-home',d+'/memory'],capture_output=True,text=True,check=True);return json.loads(r.stdout) if '--json' in args else r.stdout
 run('init');run('learn-path','--path',str(root))
 out={}
 for v in p['queries']:
  out[v['id']]={m:run('context','--query',v['query'],*flags,'--json') for m,flags in [('raw',[]),('compact',['--compact'])]}
 (D/'independent.json').write_text(json.dumps(out,indent=2))
