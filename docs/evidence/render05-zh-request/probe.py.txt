import argparse,json,subprocess,tempfile,hashlib
from pathlib import Path
ap=argparse.ArgumentParser();ap.add_argument('--label',required=True);args=ap.parse_args()
D=Path('docs/evidence/render05-zh-request');p=json.loads((D/'protocol.json').read_text())
out={}
for variant,sources in [('consumed',p['sources']),('discarded',p['discarded_sources'])]:
 with tempfile.TemporaryDirectory()as d:
  root=Path(d)/'source';root.mkdir()
  for f,s in sources.items():(root/f).write_text(s)
  def run(*args):
   r=subprocess.run(['python','tools/agent_memory.py',*args,'--project',str(root),'--memory-home',d+'/memory'],capture_output=True,text=True,check=True)
   return json.loads(r.stdout)if'--json'in args else r.stdout
  run('init');run('learn-path','--path',str(root))
  queries=p['queries']if variant=='consumed'else p['queries'][:1]
  out[variant]={q['id']:{mode:run('context','--query',q['query'],*flags,'--json')for mode,flags in [('raw',[]),('compact',['--compact'])]}for q in queries}
(D/(args.label+'.json')).write_text(json.dumps(out,ensure_ascii=False,indent=2)+'\n')
for family,records in out.items():
 for name,record in records.items():
  raw=record['raw']['query_handoff'];comp=record['compact']['query_handoff']
  print(family,name,'anchors',[a['file_path']for a in comp.get('code_anchors',[])],'logs',len(raw.get('log_anchors',[])),'path',raw.get('path_context',{}).get('activated'))
