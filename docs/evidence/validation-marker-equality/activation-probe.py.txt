import json,tempfile,subprocess
from pathlib import Path

with tempfile.TemporaryDirectory()as d:
 root=Path(d)/'source';root.mkdir()
 for f,s in p['sources'].items():(root/f).write_text(s)
 def run(script,*args):

 def snap(script):return {r['qualified_name']:r for r in run(script,'list','--type','code-symbol','--limit','100','--json')if r.get('qualified_name')}
 run(old,'init');run(old,'learn-path','--path',str(root));out={'before':snap(old),'code_only':snap(new)};run(new,'learn-path','--path',str(root));out['after_merge']=snap(new)

  assert 'statewrite'in out['before'][name]['method_evidence'].split()

  assert 'statewrite'not in out['after_merge'][name]['method_evidence'].split()
  assert 'stateread'in out['after_merge'][name]['method_evidence'].split()

 for name,row in out['before'].items():
  for key in ['file_path','qualified_name','symbol_key','start_line','end_line','source_digest','string_evidence']:assert row.get(key)==out['after_merge'][name].get(key),(name,key)
 (D/'activation.json').write_text(json.dumps(out,indent=2));print('two comparison write hints removed only after merge; six true write controls preserved')
