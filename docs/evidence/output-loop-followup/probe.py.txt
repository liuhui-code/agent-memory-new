import json, subprocess, tempfile
from pathlib import Path
p=Path('docs/evidence/output-loop-followup'); protocol=json.loads((p/'protocol.json').read_text())
with tempfile.TemporaryDirectory() as d:
 project=Path(d)
 for name,source in protocol['sources'].items():(project/name).write_text(source)
 def run(*args):
  r=subprocess.run(['python','tools/agent_memory.py',*args,'--project',d],capture_output=True,text=True,check=True)
  return json.loads(r.stdout) if '--json' in args else r.stdout
 run('init');run('learn-path','--path',d)
 result=[]
 for case in protocol['queries']:
  raw=run('context','--query',case['query'],'--json')
  compact=run('context','--query',case['query'],'--compact','--json')
  result.append({'case':case,'raw':raw,'compact':compact})
 (p/'independent-baseline.json').write_text(json.dumps(result,indent=2))
 for row in result: print(row['case']['query'],[(a['file_path'],a.get('symbol')) for a in row['compact']['query_handoff']['code_anchors']])
