import json,tempfile,subprocess,os
from pathlib import Path
D=Path('docs/evidence/validation-mechanism');p=json.loads((D/'supported-protocol.json').read_text())
with tempfile.TemporaryDirectory() as d:
 root=Path(d)/'source';root.mkdir()
 for f,s in p['sources'].items():(root/f).write_text(s)
 def run(*args):
  r=subprocess.run(['python','tools/agent_memory.py',*args,'--project',str(root),'--memory-home',d+'/memory'],capture_output=True,text=True,check=True);return json.loads(r.stdout)if '--json'in args else r.stdout
 run('init');learn=run('learn-path','--path',str(root),'--json')
 out={'learn':learn,'symbols':run('list','--type','code-symbol','--limit','100','--json'),'contexts':{q:run('context','--query',q,'--json')for q in p['queries']}}
 (D/(os.environ.get('OUT','supported-before')+'.json')).write_text(json.dumps(out,indent=2))
 print([(r['qualified_name'],r['method_evidence'],r['start_line'],r['end_line'])for r in out['symbols']if r.get('symbol_type')=='function'])
