import json,subprocess,tempfile
from pathlib import Path
p=Path('docs/evidence/causal-log-followup');protocol=json.loads((p/'protocol.json').read_text())
with tempfile.TemporaryDirectory() as d:
 root=Path(d)
 for name,source in protocol['sources'].items():(root/name).write_text(source)
 def run(*args):
  r=subprocess.run(['python','tools/agent_memory.py',*args,'--project',d,'--memory-home',d+'/memory'],capture_output=True,text=True,check=True);return json.loads(r.stdout) if '--json' in args else r.stdout
 run('init');run('learn-path','--path',d)
 raw=run('context','--query',protocol['query'],'--json');compact=run('context','--query',protocol['query'],'--compact','--json')
 (p/'independent-baseline.json').write_text(json.dumps({'raw':raw,'compact':compact},indent=2))
 for name,row in [('raw',raw),('compact',compact)]:print(name,[a['entry'] for a in row['query_handoff']['path_context']['path_candidates']],row.get('output_budget'))
