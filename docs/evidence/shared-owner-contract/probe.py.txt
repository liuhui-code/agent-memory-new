import json,tempfile,subprocess,os
from pathlib import Path
repo=Path('/workspace/scratch/dcdb41e16594/recovery-current/agent-memory-new');out=Path('/workspace/scratch/dcdb41e16594/shared-owner-contract-probe')
protocol=json.loads((out/'protocol.json').read_text());results=[]
for case in protocol['cases']:
 with tempfile.TemporaryDirectory() as d:
  root=Path(d)
  for name,source in case['sources'].items():(root/name).write_text(source)
  def run(*args):
   r=subprocess.run(['python',str(repo/'tools/agent_memory.py'),*args,'--project',d,'--memory-home',d+'/mem'],env={**os.environ,'PYTHONDONTWRITEBYTECODE':'1'},capture_output=True,text=True,check=True)
   return json.loads(r.stdout) if '--json' in args else r.stdout
  run('init');run('learn-path','--path','.')
  results.append({'id':case['id'],'query':case['query'],'symbols':run('list','--type','code-symbol','--json'),'raw':run('context','--query',case['query'],'--json'),'compact':run('context','--query',case['query'],'--compact','--json')})
(out/'results.json').write_text(json.dumps(results,indent=2))
